cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(vsweep C CXX ASM)
pico_sdk_init()
add_executable(vsweep main.c oc.c)
target_include_directories(vsweep PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(vsweep pico_stdlib hardware_clocks hardware_vreg hardware_adc hardware_watchdog)
target_compile_options(vsweep PRIVATE -O2)
pico_enable_stdio_usb(vsweep 1)
pico_enable_stdio_uart(vsweep 0)
pico_add_extra_outputs(vsweep)
